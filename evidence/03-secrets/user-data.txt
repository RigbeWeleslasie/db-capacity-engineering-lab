#!/bin/bash
# =============================================================================
# user-data — install nginx, render the readiness-gated proxy, start the app.
#
# SECURITY: we pass the secret ARN + DB endpoint, NEVER the secret value. On
# real EC2 this file is readable via IMDS; on LocalStack it sits world-readable
# at /var/lib/cloud/instances/<id>/. Same discipline either way: no plaintext
# credential here. The app calls GetSecretValue at boot using DB_SECRET_ARN.
# =============================================================================
set -euxo pipefail
export DEBIAN_FRONTEND=noninteractive

# --- nginx --------------------------------------------------------------------
apt-get update
apt-get install -y --no-install-recommends nginx curl

cat > /etc/nginx/sites-available/app.conf <<'NGINXCONF'
(nginx.conf.tftpl rendered separately -- no secret content, omitted here for brevity)
NGINXCONF
ln -sf /etc/nginx/sites-available/app.conf /etc/nginx/sites-enabled/default
rm -f /etc/nginx/sites-enabled/default.dpkg-dist 2>/dev/null || true
nginx -t
(systemctl restart nginx || service nginx restart || nginx) 2>/dev/null || nginx

# --- app ----------------------------------------------------------------------
# Creds are resolved by the app from Secrets Manager at boot. We export only the
# ARN + endpoint + non-secret service config.
cd "/usr/src/app"

export AWS_REGION="us-east-1"
export AWS_DEFAULT_REGION="us-east-1"
# LocalStack target only: point the SDK at the emulator and use its dummy creds
# (account 000000000000 — not real secrets). Gated so the SAME image runs against
# real AWS by passing aws_endpoint_url = "": no endpoint override and no static
# creds, so the SDK falls through to the instance-profile role. This gate lives in
# infra glue, not application code — the app itself has no isLocalStack branch.
export AWS_ENDPOINT_URL="http://localhost.localstack.cloud:4566"
export AWS_ACCESS_KEY_ID="test"
export AWS_SECRET_ACCESS_KEY="test"
export DB_SECRET_ARN="arn:aws:secretsmanager:us-east-1:000000000000:secret:regional-health/db-xbIZvk"
export DB_HOST="mysql-2343e8b8-rigbewweleslasie-2c2e.i.aivencloud.com"
export DB_PORT="16437"
